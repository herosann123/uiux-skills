Particle Network

Connected.

鼠标或手指靠近粒子会被推开，形成动态交互网络。

// 核心：粒子类 + 鼠标排斥 + 邻近连线
class Particle {
  update(w, h) {
    this.x += this.vx;
    this.y += this.vy;
    const dx = this.x - mouse.x;
    const dy = this.y - mouse.y;
    const d = Math.sqrt(dx*dx + dy*dy);
    if (d < 120) {
      const force = (120 - d) / 120;
      this.x += (dx/d) * force * 3;
      this.y += (dy/d) * force * 3;
    }
  }
}

Micro-interaction · 触觉反馈

水波纹涟漪

点击位置发出一个从触点扩散的圆环。本质是动态插入一个 position:absolute 的小圆，transform: scale() + opacity 同时过渡。

demo · click buttons
Click any button · 点击任意按钮
// 核心：点击时插入 span + CSS 动画 + 自动清理
btn.addEventListener('click', (e) => {
  const rect = btn.getBoundingClientRect();
  const ripple = document.createElement('span');
  ripple.className = 'ripple';
  ripple.style.left = (e.clientX - rect.left) + 'px';
  ripple.style.top  = (e.clientY - rect.top)  + 'px';
  btn.appendChild(ripple);
  setTimeout(() => ripple.remove(), 800);
});

// F.L.I.P. — 列表 → 详情 无缝过渡
function open(card) {
  const first = card.getBoundingClientRect();   // First
  inner.style.left = first.left + 'px';
  inner.style.top  = first.top  + 'px';
  inner.style.width  = first.width  + 'px';
  inner.style.height = first.height + 'px';
  detail.classList.add('open');
  requestAnimationFrame(() => {                   // Last → Play
    inner.style.transition = 'all 0.6s cubic-bezier(0.4,0,0.2,1)';
    inner.style.left = '0'; inner.style.top = '0';
    inner.style.width = '100%'; inner.style.height = '100%';
  });
}

// 核心：递归 setTimeout + 可变延迟 + 标点慢一拍
function typeWriter(text, el, speed = 'normal') {
  let i = 0;
  const ranges = { fast: [10,30], normal: [20,80], slow: [80,200] };
  const [min, max] = ranges[speed];
  function tick() {
    if (i >= text.length) return;
    el.textContent += text[i++];
    let delay = min + Math.random() * (max - min);
    if ('，。？！'.includes(text[i-1])) delay = max * 2;
    setTimeout(tick, delay);
  }
  tick();
}
