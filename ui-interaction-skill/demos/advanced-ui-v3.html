自定义光标

隐藏原生光标，绘制双层光标：内点立即跟随，外圈带阻尼惯性。悬停可交互元素时外圈放大、变色，常用于 Linear / Vercel / Stripe 等高端产品页。

Touch. Feel.

当鼠标移过可交互元素时光标会变形 —— 让交互的「起点」本身就是视觉语言的一部分。

Hover me
↳ 双层光标 · 惯性延迟 · 悬停放大 ↳ 触摸设备自动禁用自定义光标,恢复原生体验 
// 核心：双层光标 + requestAnimationFrame 惯性跟随
const dot = document.querySelector('.cursor-dot');
const ring = document.querySelector('.cursor-ring');
let mx = 0, my = 0, rx = 0, ry = 0;

document.addEventListener('mousemove', (e) => {
  mx = e.clientX; my = e.clientY;
  dot.style.transform = `translate(${mx}px, ${my}px) translate(-50%,-50%)`;
});

(function tick() {
  rx += (mx - rx) * 0.18;        // 阻尼
  ry += (my - ry) * 0.18;
  ring.style.transform = `translate(${rx}px, ${ry}px) translate(-50%,-50%)`;
  requestAnimationFrame(tick);
})();

// 悬停放大
document.querySelectorAll('.hover-target').forEach(el => {
  el.addEventListener('mouseenter', () => ring.classList.add('hover'));
  el.addEventListener('mouseleave', () => ring.classList.remove('hover'));
});

噪点纹理

在背景上叠加一层细密噪点，瞬间让平面背景产生「印刷品 / 胶片」质感。SVG feTurbulence 生成噪声 + 短间隔 steps() 抖动 = "expensive" 视觉的秘诀。

demo · animated grain
Animated Noise

Premium Grain.

每一帧重新生成噪点图案 —— 让深色背景瞬间告别「塑料感」。

// 核心：feTurbulence 噪声 + 每帧重绘
function renderNoise() {
  const ctx = canvas.getContext('2d');
  const img = ctx.createImageData(w, h);
  for (let i = 0; i < img.data.length; i += 4) {
    const v = Math.random() * 255;
    img.data[i] = img.data[i+1] = img.data[i+2] = v;
    img.data[i+3] = 255;
  }
  ctx.putImageData(img, 0, 0);
  requestAnimationFrame(renderNoise);
}

滚动驱动数字

把滚动进度 0~1 映射到数字 0~N，数字随用户滚动「滚出来」。Apple 产品页典型手法：用 ScrollTrigger 监听 progress → 平滑 lerp → tabular-nums 数字无跳动。

// 核心：ScrollTrigger.progress → 平滑数字
const numEl = document.getElementById('scrollNumber');
ScrollTrigger.create({
  trigger: track,
  start: "top top",
  end: "bottom bottom",
  onUpdate: (self) => {
    const target = Math.round(self.progress * 240);
    current += (target - current) * 0.1;     // 平滑
    numEl.textContent = Math.round(current);
    bar.style.width = (self.progress * 100) + '%';
  }
});

// 模糊匹配：高亮 query 在 title 中的字符
function highlight(text, q) {
  const re = new RegExp(`(${q})`, 'gi');
  return text.replace(re, '<mark>$1</mark>');
}

// 核心：同位多卡 + data-stack-pos 阶梯偏移 + 顶卡抽拉
// [0] = 当前激活卡；每张按 i 写入 stack-pos，CSS 用属性选择器控样式
const cards = Array.from(deck.querySelectorAll('.drawer-card'));
const order = cards.slice();

function applyStack() {
  order.forEach((c, i) => {
    if (i === 0) { c.classList.add('active'); c.removeAttribute('data-stack-pos'); }
    else          { c.classList.remove('active'); c.dataset.stackPos = i; }
  });
}

function next()  { order.push(order.shift()); applyStack(); }
function prev()  { order.unshift(order.pop()); applyStack(); }
function goTo(c) {                       // 书签点击
  const i = order.indexOf(c); if (i <= 0) return;
  order.splice(i, 1); order.unshift(c); applyStack();
}

// wheel — 桌面端（必须 preventDefault，否则页面跟着滚）
let lock = false;
window.addEventListener('wheel', (e) => {
  e.preventDefault();
  if (lock || Math.abs(e.deltaY) < 4) return;
  lock = true; setTimeout(() => lock = false, 200);
  e.deltaY > 0 ? next() : prev();
}, { passive: false });

// touch — 触摸设备 (≥40px 阈值)
let ty = null;
window.addEventListener('touchstart', (e) => { ty = e.touches[0].clientY; }, { passive: true });
window.addEventListener('touchend',   (e) => {
  if (ty === null) return;
  const dy = ty - e.changedTouches[0].clientY;
  if (Math.abs(dy) > 40) dy > 0 ? next() : prev();
  ty = null;
}, { passive: true });

// keyboard — ↑↓←→
window.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowDown' || e.key === 'ArrowRight') { e.preventDefault(); next(); }
  if (e.key === 'ArrowUp'   || e.key === 'ArrowLeft')  { e.preventDefault(); prev(); }
});

applyStack();
